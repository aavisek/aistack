## Why this matters
An API is easier to change when HTTP routing, business rules, and stored data have separate homes. Express provides routing tools, while TypeScript checks that the parts agree.

## What you will build
You will create an in-memory books API with list filtering and pagination, lookup, create, replace, patch, and delete routes. Tests exercise every endpoint and missing-resource behavior.

## Step 2: Keep data and book rules in the service
The service owns the array and mutations. IDs increase for the life of this process; stopping and restarting intentionally resets the sample data.

```ts title="src/books.service.ts" file
export type Book = {
  id: number;
  title: string;
  author: string;
  year: number;
};

export type NewBook = Omit<Book, "id">;

let nextId = 4;
const books: Book[] = [
  { id: 1, title: "The Hobbit", author: "J.R.R. Tolkien", year: 1937 },
  { id: 2, title: "Kindred", author: "Octavia E. Butler", year: 1979 },
  { id: 3, title: "The Left Hand of Darkness", author: "Ursula K. Le Guin", year: 1969 }
];

export const booksService = {
  list(author: string, page: number, limit: number): Book[] {
    const matching = books.filter((book) =>
      book.author.toLowerCase().includes(author.toLowerCase()));
    const offset = (page - 1) * limit;
    return matching.slice(offset, offset + limit);
  },
  find(id: number): Book | undefined {
    return books.find((book) => book.id === id);
  },
  create(input: NewBook): Book {
    const book = { id: nextId++, ...input };
    books.push(book);
    return book;
  },
  replace(id: number, input: NewBook): Book | undefined {
    const index = books.findIndex((book) => book.id === id);
    if (index < 0) return undefined;
    books[index] = { id, ...input };
    return books[index];
  },
  patch(id: number, input: Partial<NewBook>): Book | undefined {
    const book = books.find((candidate) => candidate.id === id);
    if (!book) return undefined;
    Object.assign(book, input);
    return book;
  },
  remove(id: number): boolean {
    const index = books.findIndex((book) => book.id === id);
    if (index < 0) return false;
    books.splice(index, 1);
    return true;
  },
  reset(): void {
    books.splice(0, books.length,
      { id: 1, title: "The Hobbit", author: "J.R.R. Tolkien", year: 1937 },
      { id: 2, title: "Kindred", author: "Octavia E. Butler", year: 1979 },
      { id: 3, title: "The Left Hand of Darkness", author: "Ursula K. Le Guin", year: 1969 });
    nextId = 4;
  }
};
```

## Step 3: Define the router and application
The router translates HTTP into service calls. A small logger records method and elapsed time; the final route handler returns JSON 404 for paths that do not exist.

```ts title="src/books.routes.ts" file
import { Router } from "express";
import { booksService, type NewBook } from "./books.service.js";

export const booksRouter = Router();
function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}
function isBookInput(value: unknown): value is NewBook {
  return isRecord(value) && typeof value.title === "string" &&
    typeof value.author === "string" && typeof value.year === "number";
}
function isBookPatch(value: unknown): value is Partial<NewBook> {
  return isRecord(value) && Object.keys(value).length > 0 &&
    (!("title" in value) || typeof value.title === "string") &&
    (!("author" in value) || typeof value.author === "string") &&
    (!("year" in value) || typeof value.year === "number");
}
const idFrom = (value: string) => Number.isInteger(Number(value)) && Number(value) > 0
  ? Number(value)
  : -1;

booksRouter.get("/", (req, res) => {
  const page = Math.max(1, Number(req.query.page) || 1);
  const limit = Math.min(100, Math.max(1, Number(req.query.limit) || 20));
  const author = typeof req.query.author === "string" ? req.query.author : "";
  res.json(booksService.list(author, page, limit));
});

booksRouter.get("/:id", (req, res) => {
  const book = booksService.find(idFrom(req.params.id));
  if (!book) return res.status(404).json({ error: "Book not found" });
  res.json(book);
});

booksRouter.post("/", (req, res) => {
  if (!isBookInput(req.body)) return res.status(400).json({ error: "Invalid book" });
  const book = booksService.create(req.body);
  res.location(`/books/${book.id}`).status(201).json(book);
});

booksRouter.put("/:id", (req, res) => {
  if (!isBookInput(req.body)) return res.status(400).json({ error: "Invalid book" });
  const book = booksService.replace(idFrom(req.params.id), req.body);
  if (!book) return res.status(404).json({ error: "Book not found" });
  res.json(book);
});

booksRouter.patch("/:id", (req, res) => {
  if (!isBookPatch(req.body)) return res.status(400).json({ error: "Invalid book patch" });
  const book = booksService.patch(idFrom(req.params.id), req.body);
  if (!book) return res.status(404).json({ error: "Book not found" });
  res.json(book);
});

booksRouter.delete("/:id", (req, res) => {
  if (!booksService.remove(idFrom(req.params.id))) {
    return res.status(404).json({ error: "Book not found" });
  }
  res.status(204).end();
});
```

```ts title="src/app.ts" file
import express from "express";
import { booksRouter } from "./books.routes.js";

export const app = express();
app.use(express.json());
app.use((req, res, next) => {
  const start = performance.now();
  res.once("finish", () => {
    console.log(`${req.method} ${req.path} ${res.statusCode} ${Math.round(performance.now() - start)}ms`);
  });
  next();
});
app.use("/books", booksRouter);
app.use((req, res) => res.status(404).json({ error: "Route not found" }));
```

## Step 4: Exercise every endpoint
Supertest can send requests directly to the Express application without opening a port. Reset the in-memory data before each test so results do not depend on test order.

```ts title="tests/books.test.ts" file
import test, { beforeEach } from "node:test";
import assert from "node:assert/strict";
import request from "supertest";
import { app } from "../src/app.js";
import { booksService } from "../src/books.service.js";

beforeEach(() => booksService.reset());

test("GET list filters and paginates", async () => {
  const response = await request(app).get("/books?author=butler&page=1&limit=1");
  assert.equal(response.status, 200);
  assert.equal(response.body.length, 1);
  assert.equal(response.body[0].id, 2);
});

test("GET one and returns 404 when absent", async () => {
  assert.equal((await request(app).get("/books/1")).body.title, "The Hobbit");
  assert.equal((await request(app).get("/books/99")).status, 404);
});

test("POST creates with Location and 201", async () => {
  const response = await request(app).post("/books")
    .send({ title: "Dawn", author: "Octavia E. Butler", year: 1987 });
  assert.equal(response.status, 201);
  assert.equal(response.headers.location, "/books/4");
});

test("PUT replaces, PATCH changes, and both return 404 when absent", async () => {
  const replacement = { title: "Dawn", author: "Octavia E. Butler", year: 1987 };
  assert.equal((await request(app).put("/books/1").send(replacement)).body.title, "Dawn");
  assert.equal((await request(app).patch("/books/1").send({ year: 1988 })).body.year, 1988);
  assert.equal((await request(app).put("/books/99").send(replacement)).status, 404);
  assert.equal((await request(app).patch("/books/99").send({ year: 1988 })).status, 404);
});

test("DELETE returns 204 and 404 on a later delete", async () => {
  assert.equal((await request(app).delete("/books/1")).status, 204);
  assert.equal((await request(app).delete("/books/1")).status, 404);
});
```

## How it works
Express matches a method and path, then runs middleware and a route handler. The router is the HTTP boundary; the service owns behavior and data. The array is temporary process memory, so a restart restores its initial seed.

## Verify against the requirements
- [ ] GET collection supports author filtering and page/limit; GET by ID returns a book or 404 → Step 3.
- [ ] POST returns 201 and `Location`; PUT replaces; PATCH partially changes; DELETE returns 204 → Steps 3–4.
- [ ] Router, service, and in-memory array have separate responsibilities; logger and JSON 404 handler are present → Steps 2–3.
- [ ] Tests cover all six methods and missing-resource paths → Step 4.
- [ ] Restarting the process resets the documented in-memory seed → How it works.

## Common errors
### TypeScript cannot find an Express type
Install `@types/express` and make sure the project compiler includes `src`.
### `Location` is missing
Set it on the create response before sending the body.
### A delete response hangs in the client
End the response with status 204 and no JSON body.
