## Why this matters
Node.js runs JavaScript outside a web browser, so you can write useful terminal tools. Modules split a program into files, and `fs/promises` and `path` help it safely work with files and folders. A small CLI is like a vending machine: you give it a command and input, and it returns a clear result.

## What you will build
You will build a `notes` command with `add`, `list`, `done`, and `remove` actions. It saves notes in `notes.json`, remembers the next ID even after deleting notes, reads an optional path from `NOTES_FILE`, and reports invalid input or damaged data without a stack trace.

```text title="Folder structure"
fd-03-notes/
├─ notes.js
├─ notes.json              created automatically
└─ src/
   ├─ commands.js
   ├─ commands.test.js
   └─ store.js
```

## Step 2: Store notes safely
`fs/promises` is Node's promise-based file module. A promise represents work that will finish later. `path` joins folder and file names using the rules of the current operating system.

Create `src/store.js` with this full content. The saved object contains both the notes and `nextId`, so removing a note never makes its number available again.

Open or create the file in VS Code by running `code src/store.js` from the project folder, or use **File → New Text File** and save it at that path.

```js title="src/store.js" file
import { readFile, writeFile } from "node:fs/promises";

export async function readState(filePath) {
  try {
    const contents = await readFile(filePath, "utf8");
    const state = JSON.parse(contents);
    if (
      !state ||
      !Array.isArray(state.notes) ||
      !Number.isSafeInteger(state.nextId) ||
      state.nextId < 1 ||
      state.notes.some((note) => !Number.isSafeInteger(note.id) || typeof note.text !== "string")
    ) {
      throw new Error("Notes file has an unexpected format.");
    }
    return state;
  } catch (error) {
    if (error.code === "ENOENT") return { nextId: 1, notes: [] };
    if (error instanceof SyntaxError) {
      throw new Error("Notes file is not valid JSON. Back it up before replacing it.");
    }
    throw error;
  }
}

export async function writeState(filePath, state) {
  await writeFile(filePath, `${JSON.stringify(state, null, 2)}\n`, "utf8");
}
```

> **Why it works:** `ENOENT` means the file does not exist yet, so the first command starts with an empty list. Invalid JSON becomes a short helpful error; it is not silently overwritten.

## Step 3: Add the note commands
`process.argv` is the list of command-line words. `process.env` contains environment settings; here `NOTES_FILE` can choose another data file. The default is `notes.json` in the current folder.

Create `src/commands.js` with the complete command handler:

```js title="src/commands.js" file
import { resolve } from "node:path";
import { readState, writeState } from "./store.js";

export function usage() {
  return [
    "Usage:",
    '  node notes.js add "text"',
    "  node notes.js list",
    "  node notes.js done <id>",
    "  node notes.js remove <id>"
  ].join("\n");
}

export async function runCommand(args, options = {}) {
  const filePath = resolve(options.filePath ?? process.env.NOTES_FILE ?? "notes.json");
  const print = options.print ?? console.log;
  const report = options.report ?? console.error;
  const [command, ...values] = args;

  try {
    if (command === "list" && values.length === 0) {
      const state = await readState(filePath);
      print(state.notes.length
        ? state.notes.map((note) => `${note.done ? "[x]" : "[ ]"} #${note.id} ${note.text}`).join("\n")
        : "No notes yet.");
      return 0;
    }

    if (command === "add" && values.length === 1 && values[0].trim()) {
      const state = await readState(filePath);
      const note = { id: state.nextId, text: values[0].trim(), done: false };
      state.notes.push(note);
      state.nextId += 1;
      await writeState(filePath, state);
      print(`Added note #${note.id}`);
      return 0;
    }

    if ((command === "done" || command === "remove") && values.length === 1 && /^\d+$/.test(values[0])) {
      const id = Number(values[0]);
      const state = await readState(filePath);
      const index = state.notes.findIndex((note) => note.id === id);
      if (index < 0) {
        report(`No note exists with ID ${id}.`);
        return 1;
      }
      if (command === "done") {
        state.notes[index].done = true;
        await writeState(filePath, state);
        print(`Completed note #${id}`);
      } else {
        state.notes.splice(index, 1);
        await writeState(filePath, state);
        print(`Removed note #${id}`);
      }
      return 0;
    }

    report(usage());
    return 1;
  } catch (error) {
    report(error instanceof Error ? error.message : "The notes command failed.");
    return 1;
  }
}
```

The handler returns `0` when a command works and `1` for invalid input or a missing note. Returning a status lets the terminal and tests tell success from failure.

## Step 4: Connect the terminal to the handler
An ES module is a JavaScript file that shares code with `import` and `export`. Add `"type": "module"` to `package.json` so Node knows these files use ES modules.

Create the `package.json` file:

```json title="package.json" file
{
  "name": "fd-03-notes",
  "version": "1.0.0",
  "private": true,
  "type": "module",
  "scripts": {
    "start": "node notes.js",
    "test": "node --test",
    "build": "node --check notes.js && node --check src/commands.js && node --check src/store.js"
  }
}
```

Create `notes.js`, the small entry point that passes arguments to the handler and sets the process exit code:

```js title="notes.js" file

const exitCode = await runCommand(process.argv.slice(2));
```

## Step 5: Add automated tests
An automated test is a small program that checks expected behavior for you. Node's built-in `node:test` module runs tests without installing any packages.

Create `src/commands.test.js` with these six complete tests. Each test gets its own temporary folder so it cannot change your real notes.

```js title="src/commands.test.js" file
import test from "node:test";
import assert from "node:assert/strict";
import { mkdtemp, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { runCommand } from "./commands.js";

async function withNotes(run) {
  const folder = await mkdtemp(join(tmpdir(), "fd-03-notes-"));
  try {
    await run(join(folder, "notes.json"));
  } finally {
    await rm(folder, { recursive: true, force: true });
  }
}

test("add writes a note and the next ID", async () => {
  await withNotes(async (filePath) => {
    const output = [];
    assert.equal(await runCommand(["add", "Buy milk"], { filePath, print: (text) => output.push(text) }), 0);
    assert.deepEqual(JSON.parse(await readFile(filePath, "utf8")), {
      nextId: 2,
      notes: [{ id: 1, text: "Buy milk", done: false }]
    });
    assert.deepEqual(output, ["Added note #1"]);
  });
});

test("list prints saved notes", async () => {
  await withNotes(async (filePath) => {
    await runCommand(["add", "Buy milk"], { filePath, print: () => {} });
    const output = [];
    await runCommand(["list"], { filePath, print: (text) => output.push(text) });
    assert.deepEqual(output, ["[ ] #1 Buy milk"]);
  });
});

test("done marks a saved note complete", async () => {
  await withNotes(async (filePath) => {
    await runCommand(["add", "Buy milk"], { filePath, print: () => {} });
    const output = [];
    await runCommand(["done", "1"], { filePath, print: (text) => output.push(text) });
    assert.deepEqual(output, ["Completed note #1"]);
  });
});

test("removed IDs are not reused", async () => {
  await withNotes(async (filePath) => {
    await runCommand(["add", "First"], { filePath, print: () => {} });
    await runCommand(["remove", "1"], { filePath, print: () => {} });
    const output = [];
    await runCommand(["add", "Second"], { filePath, print: (text) => output.push(text) });
    assert.deepEqual(output, ["Added note #2"]);
  });
});

test("invalid input prints usage and returns a failing status", async () => {
  await withNotes(async (filePath) => {
    const errors = [];
    assert.equal(await runCommand(["unknown"], { filePath, report: (text) => errors.push(text) }), 1);
    assert.match(errors[0], /Usage:/);
  });
});

test("corrupt JSON is reported without a stack trace", async () => {
  await withNotes(async (filePath) => {
    await writeFile(filePath, "{broken");
    const errors = [];
    assert.equal(await runCommand(["list"], { filePath, report: (text) => errors.push(text) }), 1);
    assert.deepEqual(errors, ["Notes file is not valid JSON. Back it up before replacing it."]);
  });
});
```

## How it works
`notes.js` receives the words you typed. `commands.js` checks the requested action, and `store.js` reads or writes JSON with Node's file tools. The saved `nextId` works like a ticket dispenser: every new note gets a fresh number.

```text title="Command flow"
Terminal arguments → notes.js → commands.js → store.js → notes.json
```

## Common errors
### `Error [ERR_MODULE_NOT_FOUND]: Cannot find module`
The import path is misspelled, or the terminal is in the wrong folder. Check that `src/commands.js` and `src/store.js` exist, then run `node notes.js list` from the project folder.

### `SyntaxError: Cannot use import statement outside a module`
Node is treating `.js` files as CommonJS. Set `"type": "module"` in `package.json` as shown in Step 4.

### `Notes file is not valid JSON. Back it up before replacing it.`
The data file has broken JSON syntax. Copy it to a backup, inspect the copy, then fix or remove the damaged file before running the command again.

### `No note exists with ID 7.`
That ID is not in the saved notes. Run `node notes.js list` and use an ID that appears there.
