## Why this matters
An API needs durable storage when data must survive restarts. A database driver can send SQL safely without an ORM, provided every user value is passed as a parameter rather than pasted into a query string.

## What you will build

## Step 3: Put all database access in a repository
Every value from a caller is passed through `$1`, `$2`, and so on. The SQL text remains fixed; never concatenate `title` or an ID into a query.

```js title="src/todo-repository.js" file
export class TodoRepository {
  constructor(pool) {
    this.pool = pool;
  }

  async list() {
    const { rows } = await this.pool.query(
      "SELECT id, title, done, created_at FROM todos ORDER BY id"
    );
    return rows;
  }

  async find(id) {
    const { rows } = await this.pool.query(
      "SELECT id, title, done, created_at FROM todos WHERE id = $1",
      [id]
    );
    return rows[0] ?? null;
  }

  async create(title) {
    const { rows } = await this.pool.query(
      "INSERT INTO todos (title) VALUES ($1) RETURNING id, title, done, created_at",
      [title]
    );
    return rows[0];
  }

  async update(id, fields) {
    const { rows } = await this.pool.query(
      "UPDATE todos SET title = COALESCE($2, title), done = COALESCE($3, done) WHERE id = $1 RETURNING id, title, done, created_at",
      [id, fields.title ?? null, fields.done ?? null]
    );
    return rows[0] ?? null;
  }

  async remove(id) {
    const { rowCount } = await this.pool.query("DELETE FROM todos WHERE id = $1", [id]);
    return rowCount === 1;
  }

  async markDoneInTransaction(id) {
    const client = await this.pool.connect();
    try {
      await client.query("BEGIN");
      const { rows } = await client.query(
        "UPDATE todos SET done = true WHERE id = $1 RETURNING id, title, done",
        [id]
      );
      if (!rows[0]) {
        await client.query("ROLLBACK");
        return null;
      }
      await client.query("COMMIT");
      return rows[0];
    } catch (error) {
      await client.query("ROLLBACK");
      throw error;
    } finally {
      client.release();
    }
  }
}
```

## Step 4: Connect Express and close the pool gracefully
The server uses the repository rather than issuing SQL in route handlers. `SIGINT` stops accepting requests, then ends the connection pool.

```js title="src/server.js" file
import express from "express";
import { pool } from "./db.js";
import { TodoRepository } from "./todo-repository.js";

const app = express();
const todos = new TodoRepository(pool);
app.use(express.json());
app.get("/health", async (_req, res) => {
  try {
    await pool.query("SELECT 1");
    res.json({ status: "ok" });
  } catch (error) {
    console.error("Database health check failed", error);
    res.status(503).json({ status: "unavailable" });
  }
});
app.get("/todos", async (_req, res, next) => {
  try { res.json(await todos.list()); } catch (error) { next(error); }
});
app.get("/todos/:id", async (req, res, next) => {
  try {
    const todo = await todos.find(Number(req.params.id));
    if (!todo) return res.status(404).json({ error: "Todo not found" });
    res.json(todo);
  } catch (error) { next(error); }
});
app.post("/todos", async (req, res, next) => {
  try { res.status(201).json(await todos.create(req.body.title)); } catch (error) { next(error); }
});
app.patch("/todos/:id", async (req, res, next) => {
  try {
    const todo = await todos.update(Number(req.params.id), req.body);
    if (!todo) return res.status(404).json({ error: "Todo not found" });
    res.json(todo);
  } catch (error) { next(error); }
});
app.delete("/todos/:id", async (req, res, next) => {
  try {
    if (!await todos.remove(Number(req.params.id))) return res.status(404).end();
    res.status(204).end();
  } catch (error) { next(error); }
});

const server = app.listen(3000, () => console.log("Todos API listening on 3000"));
const shutdown = () => server.close(async () => {
  await pool.end();
  process.exit(0);
});
process.once("SIGINT", shutdown);
process.once("SIGTERM", shutdown);
```

## Step 5: Compare unsafe SQL with the parameterized fix
This test-only vulnerable query is intentionally isolated and read-only. The payload changes its WHERE expression and returns all rows; the repository treats the same payload as an ordinary title and returns no match.

```js title="tests/sql-injection.test.js" file
import test from "node:test";
import assert from "node:assert/strict";
import { pool } from "../src/db.js";
import { TodoRepository } from "../src/todo-repository.js";

test("parameterized values remain data, not SQL", async (t) => {
  t.after(() => pool.end());
  const injection = "' OR TRUE --";
  const vulnerableSql = `SELECT id, title FROM todos WHERE title = '${injection}'`;
  const vulnerable = await pool.query(vulnerableSql);
  assert.ok(vulnerable.rows.length >= 2);

  const repository = new TodoRepository(pool);
  assert.equal(await repository.find(injection), null);
});
```

```text title="Expected output"
The vulnerable test query returns the seeded rows.
The parameterized repository lookup returns null.
```

## How it works
`pg.Pool` reuses database connections. A repository centralizes SQL, and placeholders separate query structure from values. A transaction uses one checked-out client from `BEGIN` through `COMMIT` or `ROLLBACK`; always release that client in `finally`.

## Common errors
### PostgreSQL rejects the connection
Check that the database is healthy and that `DATABASE_URL` matches the local port and `.env` credentials.
### Data disappears after a container restart
Confirm Compose attached the named volume and avoid `docker compose down -v`.
### Injection payload changes the query
Never concatenate caller values into SQL; pass them in the query's values array.

## Recap
- Keep database values parameterized and isolate SQL in a repository.
- Use a named volume to preserve PostgreSQL data across container recreation.
- Use a single client for a transaction and release it even when an operation fails.
