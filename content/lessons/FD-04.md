## Why this matters
Node can wait for files, timers, and network responses without freezing the whole program. Asynchronous patterns are different ways to describe what should happen after that work finishes.

## Step 2: Compare the three styles
The callback version nests completions; the promise version chains them; `async/await` makes the same sequence read top to bottom. Each version sums the same three files.

```js title="async.js" file
import { readFile } from "node:fs";
import { readFile as readFilePromise } from "node:fs/promises";
import { setTimeout as delay } from "node:timers/promises";
import { pathToFileURL } from "node:url";

const files = ["data/a.txt", "data/b.txt", "data/c.txt"];
const number = (text) => Number(text.trim());

export function sumWithCallbacks(paths, done) {
  readFile(paths[0], "utf8", (error, first) => {
    if (error) return done(error);
    readFile(paths[1], "utf8", (secondError, second) => {
      if (secondError) return done(secondError);
      readFile(paths[2], "utf8", (thirdError, third) => {
        if (thirdError) return done(thirdError);
        done(null, number(first) + number(second) + number(third));
      });
    });
  });
}

export function sumWithPromises(paths) {
  return readFilePromise(paths[0], "utf8")
    .then((first) => readFilePromise(paths[1], "utf8").then((second) => [first, second]))
    .then(([first, second]) => readFilePromise(paths[2], "utf8").then((third) =>
      number(first) + number(second) + number(third)));
}

export async function sumWithAsync(paths) {
  const first = await readFilePromise(paths[0], "utf8");
  const second = await readFilePromise(paths[1], "utf8");
  const third = await readFilePromise(paths[2], "utf8");
  return number(first) + number(second) + number(third);
}

export async function sumInParallel(paths) {
  const texts = await Promise.all(paths.map((path) => readFilePromise(path, "utf8")));
  return texts.reduce((total, text) => total + number(text), 0);
}

export function sleep(ms) {
  return delay(ms);
}

export async function retry(fn, times) {
  if (!Number.isInteger(times) || times < 1) throw new RangeError("times must be a positive integer");
  let lastError;
  for (let attempt = 1; attempt <= times; attempt += 1) {
    try {
      return await fn();
    } catch (error) {
      lastError = error;
    }
  }
  throw lastError;
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const callbackSum = await new Promise((resolve, reject) =>
    sumWithCallbacks(files, (error, value) => error ? reject(error) : resolve(value)));
  console.log("callbacks:", callbackSum);
  console.log("promises:", await sumWithPromises(files));
  console.log("async/await:", await sumWithAsync(files));
}
```

## Step 3: Measure sequential and parallel work
`Promise.all` starts all independent reads before waiting. A short delay makes the timing difference visible in a fast local example; both strategies still sum file contents.

```js title="Add to async.js"
export async function compareTimings(paths) {
  const readSlowly = async (path) => {
    await sleep(150);
    return readFilePromise(path, "utf8");
  };
  const startedSequential = performance.now();
  const sequential = [];
  for (const path of paths) sequential.push(await readSlowly(path));
  const sequentialMs = performance.now() - startedSequential;

  const startedParallel = performance.now();
  const parallel = await Promise.all(paths.map(readSlowly));
  const parallelMs = performance.now() - startedParallel;
  return {
    sequentialTotal: sequential.reduce((sum, text) => sum + number(text), 0),
    parallelTotal: parallel.reduce((sum, text) => sum + number(text), 0),
    sequentialMs: Math.round(sequentialMs),
    parallelMs: Math.round(parallelMs)
  };
}
```

```text title="Expected output"
Both totals are 18; sequential time is about 450 ms and parallel time about 150 ms.
```

```bash title="Terminal" verify="manual" manual review required 2026-10-18 Linux

```

## Step 4: Predict and inspect the event loop
Predict the order before running this example. The timer and immediate are scheduled from an I/O callback so their ordering is stable; `nextTick` and promise callbacks run before that callback returns to the event loop.

```js title="event-loop.js" file
import { readFile } from "node:fs";

console.log("start");
readFile("data/a.txt", () => {
  console.log("I/O callback");
  setTimeout(() => console.log("timer"), 0);
  setImmediate(() => console.log("immediate"));
  process.nextTick(() => console.log("nextTick"));
  Promise.resolve().then(() => console.log("promise"));
});
console.log("scheduled");
```

## Step 5: Handle failures deliberately
Use `try/catch` around awaited promises and `.catch()` on a chain. `allSettled` reports each outcome without rejecting the whole group. Do not create an unhandled rejection: attach a handler immediately.

```js title="Add to async.js"
export async function showFailurePatterns() {
  try {
    await readFilePromise("data/missing.txt", "utf8");
  } catch (error) {
    console.log("await caught:", error.code);
  }

  await readFilePromise("data/missing.txt", "utf8")
    .catch((error) => console.log("chain caught:", error.code));

  const results = await Promise.allSettled([
    readFilePromise("data/a.txt", "utf8"),
    readFilePromise("data/missing.txt", "utf8")
  ]);
  console.log(results.map((result) => result.status).join(", "));
}
```

## Step 6: Test sleep and retry
Run the tests with `node --test`. The retry case succeeds on attempt three; the second test proves it stops at the limit and rethrows the last failure.

```js title="async.test.js" file
import test from "node:test";
import assert from "node:assert/strict";
import { retry, sleep, sumWithAsync, sumInParallel } from "./async.js";

test("sleep waits at least the requested interval", async () => {
  const start = performance.now();
  await sleep(20);
  assert.ok(performance.now() - start >= 15);
});

test("file sum works sequentially and in parallel", async () => {
  assert.equal(await sumWithAsync(["data/a.txt", "data/b.txt", "data/c.txt"]), 18);
  assert.equal(await sumInParallel(["data/a.txt", "data/b.txt", "data/c.txt"]), 18);
});

test("retry succeeds on its third attempt", async () => {
  let attempts = 0;
  const result = await retry(async () => {
    attempts += 1;
    if (attempts < 3) throw new Error("try again");
    return "ready";
  }, 3);
  assert.equal(result, "ready");
  assert.equal(attempts, 3);
});

test("retry fails after its limit", async () => {
  let attempts = 0;
  await assert.rejects(retry(async () => {
    attempts += 1;
    throw new Error("still unavailable");
  }, 2), /still unavailable/);
  assert.equal(attempts, 2);
});
```

## How it works
An asynchronous operation yields control while Node waits. A callback continues later; a promise represents a future result; `await` pauses only the current async function. `Promise.all` overlaps independent work, while `allSettled` preserves both successes and failures.

## Common errors
### A callback is called twice
Return immediately after handling an error so the success path cannot run too.
### A promise rejection disappears
Return or await the promise and attach a catch handler at the boundary.
### Parallel work is not faster
Only use `Promise.all` for independent tasks; dependent steps must remain sequential.

## Recap
- Callbacks, promise chains, and `async/await` can express the same asynchronous task.
- `Promise.all` overlaps independent operations; sequential work waits at each step.
