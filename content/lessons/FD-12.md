## Why this matters
Redis is a fast in-memory data store that supports more than simple key/value strings. A short-lived cache can reduce repeated database work, as long as writes invalidate cached values and the database remains the source of truth.

## What you will build
You will practice Redis strings, lists, hashes, sets, and sorted sets; then add a 60-second cache-aside read with hit/miss headers, invalidation, timing logs, and database fallback.

> **Source note:** FD-10 builds a `todos` table, not a `books` table. This lesson applies the requested get-by-ID cache pattern to `getTodoById`; use the same pattern for books if your database has that table.

## Step 2: Add a 60-second cache-aside helper
The cache is checked first. A miss reads PostgreSQL and stores the JSON for 60 seconds. Redis failures are logged and reads continue against the database.

```js title="src/todo-cache.js" file
const cacheKey = (id) => `todos:${id}`;

export async function getCachedTodo(repository, redis, id, logger = console) {
  const started = performance.now();
  const key = cacheKey(id);
  let cached;

  try {
    cached = await redis.get(key);
  } catch (error) {
    logger.warn("Redis read failed; reading PostgreSQL", error);
  }

  if (cached !== null && cached !== undefined) {
    try {
      return {
        todo: JSON.parse(cached),
        cacheStatus: "HIT",
        elapsedMs: Math.round(performance.now() - started)
      };
    } catch (error) {
      logger.warn("Invalid cached todo; reading PostgreSQL", error);
      try {
        await redis.del(key);
      } catch (deleteError) {
        logger.warn("Could not remove invalid cache value", deleteError);
      }
    }
  }

  const todo = await repository.find(id);
  if (todo) {
    try {
      await redis.set(key, JSON.stringify(todo), "EX", 60);
    } catch (error) {
      logger.warn("Redis write failed; database result is still available", error);
    }
  }
  return {
    todo,
    cacheStatus: "MISS",
    elapsedMs: Math.round(performance.now() - started)
  };
}

export async function invalidateTodo(redis, id) {
  await redis.del(cacheKey(id));
}
```

## Step 3: Set the response header and invalidate writes
`X-Cache` tells a caller whether the response came from Redis. Delete the key before changing the database; if invalidation fails, fail the write rather than leave an old value available from cache.

```js title="src/server.js route integration example"
import Redis from "ioredis";
import { getCachedTodo, invalidateTodo } from "./todo-cache.js";

const redis = new Redis(process.env.REDIS_URL ?? "redis://localhost:6379", {
  maxRetriesPerRequest: 1,
  retryStrategy: (attempt) => Math.min(attempt * 100, 1000)
});
redis.on("error", (error) => console.warn("Redis connection error", error));

app.get("/todos/:id", async (req, res, next) => {
  try {
    const result = await getCachedTodo(todos, redis, Number(req.params.id));
    res.setHeader("X-Cache", result.cacheStatus);
    console.info(`todo:${req.params.id} ${result.cacheStatus} ${result.elapsedMs}ms`);
    if (!result.todo) return res.status(404).json({ error: "Todo not found" });
    res.json(result.todo);
  } catch (error) { next(error); }
});

app.patch("/todos/:id", async (req, res, next) => {
  try {
    const id = Number(req.params.id);
    await invalidateTodo(redis, id);
    const todo = await todos.update(id, req.body);
    if (!todo) return res.status(404).json({ error: "Todo not found" });
    res.json(todo);
  } catch (error) { next(error); }
});

app.delete("/todos/:id", async (req, res, next) => {
  try {
    const id = Number(req.params.id);
    await invalidateTodo(redis, id);
    if (!await todos.remove(id)) return res.status(404).end();
    res.status(204).end();
  } catch (error) { next(error); }
});
```

## Step 4: Test misses, hits, invalidation, and Redis failure
These unit tests use small fakes, so they do not need a running Redis or PostgreSQL instance. A second identical read should use the cached copy; after an update and delete, the next read must not return old data.

```js title="tests/todo-cache.test.js" file
import test from "node:test";
import assert from "node:assert/strict";
import { getCachedTodo, invalidateTodo } from "../src/todo-cache.js";

function setup() {
  const data = new Map([[1, { id: 1, title: "First", done: false }]]);
  const cache = new Map();
  const repository = {
    async find(id) { return data.get(id) ?? null; },
    async update(id, fields) {
      const old = data.get(id);
      if (!old) return null;
      const next = { ...old, ...fields };
      data.set(id, next);
      return next;
    },
    async remove(id) { return data.delete(id); }
  };
  const redis = {
    async get(key) { return cache.get(key) ?? null; },
    async set(key, value, mode, seconds) {
      assert.equal(mode, "EX");
      assert.equal(seconds, 60);
      cache.set(key, value);
    },
    async del(key) { cache.delete(key); }
  };
  return { data, cache, repository, redis };
}

const quietLogger = { warn() {} };

test("first read misses and second read hits", async () => {
  const setupData = setup();
  assert.equal((await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger)).cacheStatus, "MISS");
  assert.equal((await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger)).cacheStatus, "HIT");
});

test("update invalidation prevents a stale read", async () => {
  const setupData = setup();
  await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger);
  await invalidateTodo(setupData.redis, 1);
  await setupData.repository.update(1, { title: "Changed" });
  const result = await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger);
  assert.equal(result.cacheStatus, "MISS");
  assert.equal(result.todo.title, "Changed");
});

test("delete invalidation prevents returning a cached record", async () => {
  const setupData = setup();
  await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger);
  await invalidateTodo(setupData.redis, 1);
  await setupData.repository.remove(1);
  assert.equal((await getCachedTodo(setupData.repository, setupData.redis, 1, quietLogger)).todo, null);
});

test("Redis errors fall back to the database", async () => {
  const setupData = setup();
  const unavailableRedis = {
    async get() { throw new Error("offline"); },
    async set() { throw new Error("offline"); },
    async del() { throw new Error("offline"); }
  };
  const result = await getCachedTodo(setupData.repository, unavailableRedis, 1, quietLogger);
  assert.equal(result.cacheStatus, "MISS");
  assert.equal(result.todo.title, "First");
});
```

## How it works
Cache-aside is application-managed: read cache, fall back to the database, then populate cache. A short TTL bounds how long a missed invalidation could matter. This sample fails writes when Redis invalidation is unavailable so it does not knowingly leave a stale cached value; production systems should also consider concurrency and durable invalidation.

## Common errors
### Every request reports MISS
Check that the Redis URL is reachable and that both requests use the same ID and key prefix.
### Updated data appears old
Invalidate the exact key after writes and verify the route calls the invalidation helper.
### Redis outage breaks reads
Catch cache connection errors on reads and query PostgreSQL as the source of truth.

## Recap
- Redis provides useful string and collection data structures, each suited to a different access pattern.
- Cache-aside reads from Redis first, falls back to PostgreSQL, and stores values with a TTL.
- Invalidate cached IDs on writes, expose HIT/MISS, and keep the database as the source of truth.
