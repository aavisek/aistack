## Why this matters
Web frameworks hide the mechanics of HTTP. A small server using Node's built-in `http` module shows how URLs, methods, headers, request bodies, and status codes fit together.

## What you will build
You will create a server with an HTML home page, JSON time and echo routes, request logging, safe body-size handling, clear 400/404/405/413 responses, and graceful shutdown.

## Step 1: Create the server
The request URL is parsed relative to a base URL. Known paths with the wrong method return 405; unknown paths return 404. The body reader enforces a 1 MiB limit while consuming the stream.

```js title="server.js" file
import { createServer } from "node:http";
import { pathToFileURL } from "node:url";

const bodyLimit = 1024 * 1024;

async function readJson(req) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > bodyLimit) {
      req.resume();
      const error = new Error("Request body exceeds 1 MiB");
      error.statusCode = 413;
      throw error;
    }
    chunks.push(chunk);
  }
  const raw = Buffer.concat(chunks).toString("utf8");
  try {
    return raw ? JSON.parse(raw) : {};
  } catch {
    const error = new Error("Request body must be valid JSON");
    error.statusCode = 400;
    throw error;
  }
}

function sendJson(res, status, value) {
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8" });
  res.end(JSON.stringify(value));
}

export function createApp() {
  return createServer(async (req, res) => {
    const started = performance.now();
    const url = new URL(req.url ?? "/", "http://localhost");
    const finish = () => console.log(
      `${req.method} ${url.pathname} ${res.statusCode} ${Math.round(performance.now() - started)}ms`
    );
    res.once("finish", finish);

    try {
      if (url.pathname === "/" && req.method === "GET") {
        res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
        res.end("<!doctype html><title>FD-05</title><h1>Plain Node server</h1>");
      } else if (url.pathname === "/api/time" && req.method === "GET") {
        sendJson(res, 200, { now: new Date().toISOString() });
      } else if (url.pathname === "/api/echo" && req.method === "GET") {
        sendJson(res, 200, { msg: url.searchParams.get("msg") ?? "" });
      } else if (url.pathname === "/api/echo" && req.method === "POST") {
        const body = await readJson(req);
        sendJson(res, 200, body);
      } else if (["/", "/api/time", "/api/echo"].includes(url.pathname)) {
        res.setHeader("Allow", url.pathname === "/api/echo" ? "GET, POST" : "GET");
        sendJson(res, 405, { error: "Method not allowed" });
      } else {
        sendJson(res, 404, { error: "Not found" });
      }
    } catch (error) {
      const status = error.statusCode ?? 400;
      if (!res.headersSent) sendJson(res, status, { error: error.message });
      else res.destroy(error);
    }
  });
}

const server = createApp();
if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  server.listen(3000, "127.0.0.1", () => console.log("Listening on http://127.0.0.1:3000"));
  process.once("SIGINT", () => server.close(() => process.exit(0)));
}
```

## Step 4: Test with Node's built-in test runner
Tests can bind the server to an ephemeral port and use `fetch`; no fixed port or external test package is needed. Cover every route and failure class before calling it done.

```js title="server.test.js" file
import test from "node:test";
import assert from "node:assert/strict";
import { createApp } from "./server.js";

test("HTTP routes and failure cases", async (t) => {
  const server = createApp();
  await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
  const address = server.address();
  const base = `http://127.0.0.1:${address.port}`;
  t.after(() => new Promise((resolve) => server.close(resolve)));

  const home = await fetch(base);
  assert.equal(home.status, 200);
  assert.match(home.headers.get("content-type"), /text\/html/);

  const time = await fetch(`${base}/api/time`);
  assert.equal(time.status, 200);
  assert.ok((await time.json()).now);

  assert.deepEqual(await (await fetch(`${base}/api/echo?msg=hi`)).json(), { msg: "hi" });
  assert.deepEqual(await (await fetch(`${base}/api/echo`, {
    method: "POST", headers: { "content-type": "application/json" }, body: '{"msg":"hi"}'
  })).json(), { msg: "hi" });

  assert.equal((await fetch(`${base}/missing`)).status, 404);
  assert.equal((await fetch(`${base}/api/time`, { method: "POST" })).status, 405);
  assert.equal((await fetch(`${base}/api/echo`, {
    method: "POST", headers: { "content-type": "application/json" }, body: "{"
  })).status, 400);
  assert.equal((await fetch(`${base}/api/echo`, {
    method: "POST", body: "x".repeat(1024 * 1024 + 1)
  })).status, 413);
  assert.equal((await fetch(`${base}/api/time`)).status, 200);
});
```

## How it works
An HTTP server receives a method, URL, headers, and optional body stream. Parse the URL once, route by method and pathname, and always answer with an explicit status and content type. Logging on the response's `finish` event records the actual status sent.

## Verify against the requirements
- [ ] GET `/`, `/api/time`, and both `/api/echo` routes respond with the correct content types → Steps 1–3.
- [ ] Malformed JSON returns 400, wrong method 405, missing route 404, and bodies over 1 MiB return 413 → Steps 1 and 3.
- [ ] Request log includes method, path, status, and elapsed milliseconds; SIGINT closes the listener → Step 1.
- [ ] At least six route and failure assertions pass using `fetch` against a started server → Step 4.

## Common errors
### The browser shows HTML for an API route
Set `Content-Type: application/json` before writing JSON.
### `req.body` is undefined
Node's built-in server does not parse request bodies; read the stream yourself.
### The server crashes on malformed JSON
Catch parse errors and send a 400 response without terminating the process.

## Recap
- Node's built-in `http` module exposes the request and response mechanics directly.
